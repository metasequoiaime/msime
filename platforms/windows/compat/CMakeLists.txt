cmake_minimum_required(VERSION 3.25)
project(MSIMEWindows7Compatibility LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Win7 Rust 兼容对象需要 MSVC 延迟加载器")
endif()
# 兼容对象与 Rust 核心使用同一静态 CRT，探针也能从产品目录外独立启动。
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_executable(windows7-import-load ../tests/runtime/windows7_import_load.cpp)
target_compile_features(windows7-import-load PRIVATE cxx_std_17)
target_compile_options(windows7-import-load PRIVATE /W4 /WX /utf-8)
set_property(TARGET windows7-import-load PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MSIME_SHERPA_INCLUDE_DIR "" CACHE PATH "固定 sherpa-onnx 源码根目录")
set(MSIME_SHERPA_RUNTIME "" CACHE FILEPATH "待验收的兼容 sherpa-onnx DLL")
if(MSIME_SHERPA_INCLUDE_DIR)
  if(NOT EXISTS "${MSIME_SHERPA_INCLUDE_DIR}/sherpa-onnx/c-api/c-api.h")
    message(FATAL_ERROR "缺少固定 sherpa-onnx C ABI 头文件")
  endif()
  add_executable(windows7-sherpa-runtime ../tests/runtime/windows7_sherpa_runtime.cpp)
  target_include_directories(windows7-sherpa-runtime PRIVATE "${MSIME_SHERPA_INCLUDE_DIR}")
  target_compile_features(windows7-sherpa-runtime PRIVATE cxx_std_17)
  target_compile_options(windows7-sherpa-runtime PRIVATE /W4 /WX /utf-8 /UNDEBUG)
  if(MSIME_SHERPA_RUNTIME)
    add_test(NAME windows7-sherpa-runtime COMMAND windows7-sherpa-runtime "${MSIME_SHERPA_RUNTIME}")
    set_tests_properties(windows7-sherpa-runtime PROPERTIES TIMEOUT 30)
  endif()
endif()
add_library(windows7-delay-hook OBJECT Windows7DelayLoad.cpp)
target_compile_features(windows7-delay-hook PRIVATE cxx_std_17)
target_compile_options(windows7-delay-hook PRIVATE /W4 /WX /utf-8)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/windows7-delay-object-$<CONFIG>.txt"
  CONTENT "$<TARGET_OBJECTS:windows7-delay-hook>")
add_executable(windows7-delay-load ../tests/runtime/windows7_delay_load.cpp $<TARGET_OBJECTS:windows7-delay-hook>)
target_compile_features(windows7-delay-load PRIVATE cxx_std_17)
target_compile_options(windows7-delay-load PRIVATE /W4 /WX /utf-8 /UNDEBUG)
target_link_libraries(windows7-delay-load PRIVATE ole32 oleaut32)
add_executable(windows7-core-load ../tests/runtime/windows7_core_load.cpp)
target_compile_features(windows7-core-load PRIVATE cxx_std_17)
target_compile_options(windows7-core-load PRIVATE /W4 /WX /utf-8 /UNDEBUG)
target_link_libraries(windows7-core-load PRIVATE ole32)
add_test(NAME windows7-delay-load COMMAND windows7-delay-load)
set_tests_properties(windows7-delay-load PROPERTIES TIMEOUT 30)
add_executable(windows7-voice-api ../tests/runtime/windows7_voice_api.cpp)
target_compile_features(windows7-voice-api PRIVATE cxx_std_17)
target_compile_options(windows7-voice-api PRIVATE /W4 /WX /utf-8 /UNDEBUG)
target_link_libraries(windows7-voice-api PRIVATE dxgi)
add_test(NAME windows7-voice-api COMMAND windows7-voice-api)
set_tests_properties(windows7-voice-api PROPERTIES TIMEOUT 30)
set(MSIME_ONNX_INCLUDE_DIR "" CACHE PATH "固定 ONNX Runtime 源码的 C ABI 头文件目录")
set(MSIME_ONNX_RUNTIME "" CACHE FILEPATH "待验收的兼容 ONNX Runtime DLL")
if(MSIME_ONNX_INCLUDE_DIR)
  if(NOT EXISTS "${MSIME_ONNX_INCLUDE_DIR}/onnxruntime_c_api.h")
    message(FATAL_ERROR "缺少固定 ONNX C ABI 头文件")
  endif()
  add_executable(windows7-onnx-session ../tests/runtime/windows7_onnx_session.cpp)
  target_include_directories(windows7-onnx-session PRIVATE "${MSIME_ONNX_INCLUDE_DIR}")
  target_compile_features(windows7-onnx-session PRIVATE cxx_std_17)
  target_compile_options(windows7-onnx-session PRIVATE /W4 /WX /utf-8 /UNDEBUG)
  if(MSIME_ONNX_RUNTIME)
    add_test(NAME windows7-onnx-session COMMAND windows7-onnx-session "${MSIME_ONNX_RUNTIME}")
    set_tests_properties(windows7-onnx-session PROPERTIES TIMEOUT 60)
  endif()
endif()
