# State the Windows this targets instead of taking whatever the toolchain
# happens to default to. The code already requires 1703: per-monitor v2 DPI
# awareness and ID2D1DeviceContext5 both arrive there, and the latter sits
# behind NTDDI_WIN10_RS2 in the Direct2D headers. Toolchains disagree on the
# default - the build worked on one MinGW and failed to find that interface on
# another - which is the kind of difference a build should not be deciding by
# accident.
add_compile_definitions(_WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000003)

  endif()

endforeach()

# Which punctuation ends the composition with the highlighted candidate. The
# set is the reference's, character for character; one missing mark is a key
# that quietly stops doing that, visible only while typing.
# Which keys edit the composition at all: letters, the manual separator, the
# Japanese long vowel, the double-pinyin ing key, Unicode digits, erase and
# caret. Dense enough that a wrong branch is an input bug and nothing else.
# 以词定字: which bare key takes the first or the last Han character out of the
# highlighted word. The direction is the feature, and the bracket pair is also
# the TSF's paging keys, so when it must not fire matters as much as when it does.

# because /WX saw the variables an elided assert had left unused, and the tests
# that still built were reporting success without checking anything. Keep the
# assertions for the test executables only - the Server, the TIP and msimeui
# keep NDEBUG, so nothing shipped gains an assert.
get_property(msime_windows_tests DIRECTORY PROPERTY TESTS)
foreach(msime_windows_test IN LISTS msime_windows_tests)
  if(TARGET ${msime_windows_test})
    target_compile_options(${msime_windows_test} PRIVATE
                           $<IF:$<CXX_COMPILER_ID:MSVC>,/UNDEBUG,-UNDEBUG>)
  endif()
endforeach()
