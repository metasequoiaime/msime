cmake_minimum_required(VERSION 3.25)
project(MSIMEWindows7Settings LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "兼容设置窗口需要 Windows 原生工具链")
endif()
include(../../Edition.cmake)
include(CTest)
file(STRINGS "../../version-win7.txt" MSIME_WINDOWS_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../version-win7.txt")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/edition.json" CONTENT "{\"edition\":\"${MSIME_EDITION}\"}\n")
find_package(nlohmann_json 3.11 REQUIRED)
set(MSIME_HOST_LIBRARY "" CACHE FILEPATH "同源 Win7 核心 DLL 的导入库")
if(NOT IS_ABSOLUTE "${MSIME_HOST_LIBRARY}" OR NOT EXISTS "${MSIME_HOST_LIBRARY}")
  message(FATAL_ERROR "MSIME_HOST_LIBRARY 必须是同源核心的绝对导入库路径")
endif()
add_library(msime-win7-settings-window STATIC SettingsWindow.cpp SettingsRoutes.cpp McpPage.cpp KeyboardSkinPage.cpp ../../../common/Telemetry.cpp ../../../common/CrashRecorder.cpp)
target_include_directories(msime-win7-settings-window PUBLIC ../../../../crates/host-api/include ../../common)
target_compile_features(msime-win7-settings-window PUBLIC cxx_std_17)
target_compile_definitions(msime-win7-settings-window PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_definitions(msime-win7-settings-window PRIVATE MSIME_WINDOWS_VERSION="${MSIME_WINDOWS_VERSION}")
target_link_libraries(msime-win7-settings-window PUBLIC nlohmann_json::nlohmann_json "${MSIME_HOST_LIBRARY}" comctl32 ole32 uuid shell32 advapi32 user32 gdi32 msimg32 crypt32 gdiplus comdlg32)
if(MSVC)
  target_compile_options(msime-win7-settings-window PRIVATE /W4 /WX /utf-8 /bigobj)
else()
  target_compile_options(msime-win7-settings-window PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(msime-client-settings-win7 WIN32 main.cpp Settings.rc)
target_compile_definitions(msime-client-settings-win7 PRIVATE MSIME_WINDOWS_VERSION="${MSIME_WINDOWS_VERSION}")
target_link_libraries(msime-client-settings-win7 PRIVATE msime-win7-settings-window)
if(MSVC)
  target_compile_options(msime-client-settings-win7 PRIVATE /W4 /WX /utf-8)
  # 清单已由 Settings.rc 嵌入，不能再让链接器生成同编号资源。
  target_link_options(msime-client-settings-win7 PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(msime-client-settings-win7 PRIVATE -municode)
endif()
if(BUILD_TESTING)
  include(../../../common/tests/capture_only.cmake)
  # 两个宿主共用的进程入口也纳入 Windows 原生设置门禁。
  add_executable(windows-settings-instance ../../tests/ui/settings_instance.cpp)
  target_compile_features(windows-settings-instance PRIVATE cxx_std_17)
  target_compile_definitions(windows-settings-instance PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(windows-settings-instance PRIVATE user32 advapi32)
  if(MSVC)
    target_compile_options(windows-settings-instance PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(windows-settings-instance PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME windows-settings-instance COMMAND windows-settings-instance)
  set_tests_properties(windows-settings-instance PROPERTIES TIMEOUT 30)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(dictionary_fixture "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  add_custom_command(OUTPUT "${dictionary_fixture}/msime-pinyin.db" "${dictionary_fixture}/msime-english.db"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/tools/make-win7-dictionary-fixture.py"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/input/fixtures/msime-phrase-backspace.sql" "${dictionary_fixture}"
    DEPENDS ../../tests/tools/make-win7-dictionary-fixture.py ../../tests/input/fixtures/msime-phrase-backspace.sql
    VERBATIM)
  add_custom_target(windows7-dictionary-fixture DEPENDS "${dictionary_fixture}/msime-pinyin.db" "${dictionary_fixture}/msime-english.db")
  foreach(suite IN ITEMS settings clipboard dictionary vocabulary plugins routes about voice keyboard keyboard-skin)
    string(REPLACE "-" "_" source_name "${suite}")
    add_executable(windows7-${suite}-test ../../tests/ui/windows7_${source_name}.cpp Settings.rc)
    target_link_libraries(windows7-${suite}-test PRIVATE msime-win7-settings-window)
    if(MSVC)
      target_compile_options(windows7-${suite}-test PRIVATE /W4 /WX /utf-8 /UNDEBUG)
      target_link_options(windows7-${suite}-test PRIVATE /MANIFEST:NO)
    else()
      target_compile_options(windows7-${suite}-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endif()
    if(suite STREQUAL "dictionary" OR suite STREQUAL "community-resources")
      add_dependencies(windows7-${suite}-test windows7-dictionary-fixture)
      add_test(NAME windows7-${suite} COMMAND windows7-${suite}-test "${dictionary_fixture}")
    else()
      add_test(NAME windows7-${suite} COMMAND windows7-${suite}-test)
    endif()
    set_tests_properties(windows7-${suite} PROPERTIES TIMEOUT 90)
  endforeach()
  # 整套设置夹具含全部页面、持久化和异步账号流程；每次 pump_until 的 15 秒断言仍保留。
  set_tests_properties(windows7-settings PROPERTIES TIMEOUT 240)
endif()
