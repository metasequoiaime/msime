cmake_minimum_required(VERSION 3.25)
project(MSIMEWindows7Panels LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Windows 7 面板需要 Windows 原生工具链")
endif()
include(../../Edition.cmake)
include(CTest)
# 同一份面板源码编成两个产品：缺省是 Win7 包的 msime-client-panels-win7.exe；打开后是现代 Windows 包的 msime-client-panels.exe（Build-Client.ps1 传 ON），版本号取 version.txt 或 Build-Client.ps1 传入的 MSIME_WINDOWS_VERSION，不链接 HarfBuzz：Win10/11 的 DirectWrite 自带表情回退，Windows7EmojiFont.h 在那里不走 HarfBuzz 整形。
option(MSIME_PANELS_MODERN_WINDOWS "编成现代 Windows 包的 msime-client-panels.exe" OFF)
if(MSIME_PANELS_MODERN_WINDOWS)
  set(msime_panels_executable msime-client-panels)
  if(NOT MSIME_WINDOWS_VERSION)
    file(STRINGS "../../version.txt" MSIME_WINDOWS_VERSION LIMIT_COUNT 1)
    string(STRIP "${MSIME_WINDOWS_VERSION}" MSIME_WINDOWS_VERSION)
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../version.txt")
else()
  set(msime_panels_executable msime-client-panels-win7)
  file(STRINGS "../../version-win7.txt" MSIME_WINDOWS_VERSION LIMIT_COUNT 1)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../version-win7.txt")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/edition.json" CONTENT "{\"edition\":\"${MSIME_EDITION}\"}\n")
find_package(nlohmann_json 3.11 REQUIRED)
if(NOT MSIME_PANELS_MODERN_WINDOWS)
  find_package(harfbuzz CONFIG REQUIRED)
endif()
set(MSIME_HOST_LIBRARY "" CACHE FILEPATH "同源核心 DLL 的导入库：Win7 包用 Win7 核心，现代包用普通核心，非 full 版本用按版本改名的那份")
if(NOT IS_ABSOLUTE "${MSIME_HOST_LIBRARY}" OR NOT EXISTS "${MSIME_HOST_LIBRARY}")
  message(FATAL_ERROR "MSIME_HOST_LIBRARY 必须是同源核心的绝对导入库路径")
endif()
add_library(msime-win7-handwriting-window STATIC HandwritingWindow.cpp ResourcePackWorker.cpp ClipboardWindow.cpp CloudClipboardWindow.cpp VocabularyWindow.cpp CloudDictionaryWindow.cpp DictionarySyncWindow.cpp EmojiWindow.cpp KeyboardWindow.cpp VoiceWindow.cpp ../../../common/Telemetry.cpp ../../../common/CrashRecorder.cpp)
target_include_directories(msime-win7-handwriting-window PUBLIC ../../../../crates/host-api/include)
target_compile_features(msime-win7-handwriting-window PUBLIC cxx_std_17)
target_compile_definitions(msime-win7-handwriting-window PUBLIC MSIME_WINDOWS_VERSION="${MSIME_WINDOWS_VERSION}")
target_compile_definitions(msime-win7-handwriting-window PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(msime-win7-handwriting-window PUBLIC nlohmann_json::nlohmann_json "${MSIME_HOST_LIBRARY}" shell32 ole32 uuid advapi32 user32 gdi32 comdlg32 comctl32 crypt32 gdiplus msimg32)
target_link_libraries(msime-win7-handwriting-window PUBLIC dwrite d2d1)
if(MSIME_PANELS_MODERN_WINDOWS)
  target_compile_definitions(msime-win7-handwriting-window PUBLIC MSIME_PANELS_MODERN_WINDOWS=1)
else()
  target_link_libraries(msime-win7-handwriting-window PUBLIC harfbuzz::harfbuzz)
  target_compile_definitions(msime-win7-handwriting-window PUBLIC MSIME_WINDOWS7_EMOJI_SHAPING=1)
endif()
if(MSVC)
  target_compile_options(msime-win7-handwriting-window PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(msime-win7-handwriting-window PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(${msime_panels_executable} WIN32 main.cpp ../../settings/win7/Settings.rc)
target_link_libraries(${msime_panels_executable} PRIVATE msime-win7-handwriting-window)
if(MSVC)
  target_compile_options(${msime_panels_executable} PRIVATE /utf-8)
  target_link_options(${msime_panels_executable} PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(${msime_panels_executable} PRIVATE -municode)
endif()
