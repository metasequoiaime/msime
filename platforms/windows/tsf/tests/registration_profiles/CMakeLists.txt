cmake_minimum_required(VERSION 3.25)
project(TsfProfileRegistrationRegression LANGUAGES CXX)
enable_testing()

# Compile the production function body, not a copied policy. Win32/COM calls
# are injected by the test below; the complete file is also cross-compiled.
set(registration_source "${CMAKE_CURRENT_SOURCE_DIR}/../../Register/Register.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${registration_source}")
file(READ "${registration_source}" registration)
string(FIND "${registration}" "BOOL RegisterProfiles()" begin)
string(FIND "${registration}" "BOOL UnregisterProfiles()" end)
if(begin LESS 0 OR end LESS_EQUAL begin)
  message(FATAL_ERROR "Could not locate production RegisterProfiles body")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${registration}" ${begin} ${length} profile_body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/registration_profiles.inc" "${profile_body}")
add_executable(msime-tsf-profile-registration-test registration_profiles.cpp)
target_include_directories(msime-tsf-profile-registration-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(msime-tsf-profile-registration-test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(msime-tsf-profile-registration-test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(msime-tsf-profile-registration-test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME msime-tsf-profile-registration COMMAND msime-tsf-profile-registration-test)
