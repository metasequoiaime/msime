# msime-engine 的 C 示例，链接 `cargo cinstall` 装好的库：
#
#   cmake -S crates/engine-capi/examples/c -B build -DMSIME_ENGINE_PREFIX=<安装前缀> -DMSIME_ENGINE_LINK=static
#
# 有 pkg-config 时按装好的 msime-engine.pc 取头文件目录和链接参数（静态链接用 `Libs.private` 里的系统库）；没有时（Windows）自己读同一个 .pc 文件的 `Libs.private`。MSIME_EXAMPLE_SYSTEM_SQLITE 打开时另外链接系统的 SQLite，核对静态库里自带的 SQLite 符号与它能共存。
cmake_minimum_required(VERSION 3.16)
project(msime_engine_example C)

set(MSIME_ENGINE_PREFIX "" CACHE PATH "cargo cinstall 的安装前缀")
set(MSIME_ENGINE_LINK "shared" CACHE STRING "shared 或 static")
option(MSIME_EXAMPLE_SYSTEM_SQLITE "同时链接系统的 SQLite" OFF)

if(NOT IS_ABSOLUTE "${MSIME_ENGINE_PREFIX}")
  message(FATAL_ERROR "MSIME_ENGINE_PREFIX 要指向 cargo cinstall 的安装前缀（绝对路径）")
endif()
if(NOT MSIME_ENGINE_LINK MATCHES "^(shared|static)$")
  message(FATAL_ERROR "MSIME_ENGINE_LINK 只能是 shared 或 static")
endif()

set(pc_file "${MSIME_ENGINE_PREFIX}/lib/pkgconfig/msime-engine.pc")
if(NOT EXISTS "${pc_file}")
  message(FATAL_ERROR "找不到 ${pc_file}")
endif()

add_executable(msime_engine_example example.c)
set_target_properties(msime_engine_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)

# `-framework X` 这样的两段参数在 CMake 的列表里被拆开了，链接前合回一项。
function(msime_join_frameworks out)
  set(result "")
  set(pending "")
  foreach(item IN LISTS ARGN)
    if(pending)
      list(APPEND result "-framework ${item}")
      set(pending "")
    elseif(item STREQUAL "-framework")
      set(pending TRUE)
    else()
      list(APPEND result "${item}")
    endif()
  endforeach()
  set(${out} "${result}" PARENT_SCOPE)
endfunction()

find_package(PkgConfig QUIET)
if(PKG_CONFIG_FOUND)
  set(ENV{PKG_CONFIG_PATH} "${MSIME_ENGINE_PREFIX}/lib/pkgconfig")
  pkg_check_modules(MSIME REQUIRED msime-engine)
  target_include_directories(msime_engine_example PRIVATE ${MSIME_INCLUDE_DIRS})
  if(MSIME_ENGINE_LINK STREQUAL "static")
    # 前缀里同时有 .a 和共享库，`-lmsime_engine` 会挑共享库，所以静态链接直接给出归档的路径，再接上 `Libs.private`。
    pkg_get_variable(msime_libdir msime-engine libdir)
    set(private ${MSIME_STATIC_LIBRARIES})
    list(REMOVE_ITEM private msime_engine)
    msime_join_frameworks(other ${MSIME_STATIC_LDFLAGS_OTHER})
    target_link_libraries(msime_engine_example PRIVATE "${msime_libdir}/libmsime_engine.a" ${private} ${other})
  else()
    target_link_libraries(msime_engine_example PRIVATE ${MSIME_LINK_LIBRARIES})
  endif()
else()
  file(STRINGS "${pc_file}" private_line REGEX "^Libs\\.private:")
  string(REGEX REPLACE "^Libs\\.private:[ ]*" "" private_line "${private_line}")
  separate_arguments(private_items UNIX_COMMAND "${private_line}")
  set(private "")
  foreach(item IN LISTS private_items)
    if(item MATCHES "^-l(.+)$")
      list(APPEND private "${CMAKE_MATCH_1}")
    else()
      list(APPEND private "${item}")
    endif()
  endforeach()
  msime_join_frameworks(private ${private})
  target_include_directories(msime_engine_example PRIVATE "${MSIME_ENGINE_PREFIX}/include")
  if(MSIME_ENGINE_LINK STREQUAL "static")
    if(MSVC)
      set(archive "${MSIME_ENGINE_PREFIX}/lib/msime_engine.lib")
    else()
      set(archive "${MSIME_ENGINE_PREFIX}/lib/libmsime_engine.a")
    endif()
    target_link_libraries(msime_engine_example PRIVATE "${archive}" ${private})
  elseif(WIN32)
    if(MSVC)
      target_link_libraries(msime_engine_example PRIVATE "${MSIME_ENGINE_PREFIX}/lib/msime_engine.dll.lib")
    else()
      target_link_libraries(msime_engine_example PRIVATE "${MSIME_ENGINE_PREFIX}/lib/msime_engine.dll.a")
    endif()
    # Windows 按可执行文件所在目录找 DLL。
    add_custom_command(TARGET msime_engine_example POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${MSIME_ENGINE_PREFIX}/bin/msime_engine.dll" "$<TARGET_FILE_DIR:msime_engine_example>")
  else()
    target_link_libraries(msime_engine_example PRIVATE "${MSIME_ENGINE_PREFIX}/lib/libmsime_engine${CMAKE_SHARED_LIBRARY_SUFFIX}")
  endif()
endif()

if(MSIME_EXAMPLE_SYSTEM_SQLITE)
  find_package(SQLite3 REQUIRED)
  target_compile_definitions(msime_engine_example PRIVATE MSIME_EXAMPLE_SYSTEM_SQLITE)
  # CMake 4.3 起导入目标叫 SQLite3::SQLite3，旧版本只有 SQLite::SQLite3。
  if(TARGET SQLite3::SQLite3)
    target_link_libraries(msime_engine_example PRIVATE SQLite3::SQLite3)
  else()
    target_link_libraries(msime_engine_example PRIVATE SQLite::SQLite3)
  endif()
endif()
